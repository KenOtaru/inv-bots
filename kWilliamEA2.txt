// Input parameters
input double   InitialLotSize = 0.01;   // Initial lot size
input double   TotalProfitTarget = 25;   // Total Profit Target in Account Currency
input double   TotalStopLoss = -200;     // Total Stop Loss in Account Currency
input int      MaxTrades = 20;          // Maximum number of simultaneous trades
input int      MagicNumber = 0123;     // Magic number for this EA
input bool     BuySignal = true;        // Enable Trading Buy Signal
input bool     SellSignal = true;        // Enable Trading Sell Signal
input double   Martingale = 2;        // Martingale multiplier
input double   nLosses = 1;            // Losses before each multiplier
input color    TextColor = clrWhite;    // Text color for chart display
input int      TextFontSize = 12;       // Font size for chart display
input bool     DebugMode = true;        // Enable debug messages

// Global variables
CTrade trade;
int g_lossesCount = 0;
int g_buyTradeCount = 0;
int g_sellTradeCount = 0; 
int g_kTradeCount = 0;
double g_totalProfit = 0;
double g_currentLotSize = InitialLotSize;
bool g_isBuyActive = false;
bool g_isSellActive = false;
datetime g_lastBarTime = 0;
double g_accumulatedLoss = 0;
double g_currentProfitTarget = TotalProfitTarget;

datetime g_lastTradeTime = 0;
datetime g_lastSignalTime = 0;

// WPR indicator handle
int g_wprHandle = INVALID_HANDLE;

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
    Print("Initializing EA with WPR...");
    
    // Create WPR indicator handle
    g_wprHandle = iCustom(_Symbol, PERIOD_CURRENT, "WPR", 80);
    if(g_wprHandle == INVALID_HANDLE)
    {
        Print("Failed to create WPR indicator handle. Error code: ", GetLastError());
        Print("Make sure WPR indicator is installed correctly.");
        return INIT_FAILED;
    }
    
    Print("EA initialized successfully!");
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Debug Print Function                                             |
//+------------------------------------------------------------------+
void DebugPrint(string message)
{
    if(DebugMode)
        Print(message);
}

//+------------------------------------------------------------------+
//| Check if a new bar has formed                                    |
//+------------------------------------------------------------------+
bool IsNewBar()
{
    datetime currentBar = iTime(_Symbol, PERIOD_CURRENT, 0);
    
    if(g_lastBarTime != currentBar)
    {
        g_lastBarTime = currentBar;
        return true;
    }
    return false;
}

//+------------------------------------------------------------------+
//| Close All Trades                                                 |
//+------------------------------------------------------------------+
void CloseAllTrades()
{
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        ulong ticket = PositionGetTicket(i);
        if(ticket > 0 && PositionSelectByTicket(ticket))
        {
            if(PositionGetString(POSITION_SYMBOL) == _Symbol && 
               PositionGetInteger(POSITION_MAGIC) == MagicNumber)
            {
                trade.PositionClose(ticket);
            }
        }
    }

    DebugPrint("All trades closed and variables reset");
}

//+------------------------------------------------------------------+
//| Calculate Total Open Trades Profit                               |
//+------------------------------------------------------------------+
double CalculateTotalProfit()
{
    double totalProfit = 0;
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        ulong ticket = PositionGetTicket(i);
        if(ticket > 0 && PositionSelectByTicket(ticket))
        {
            if(PositionGetString(POSITION_SYMBOL) == _Symbol && 
               PositionGetInteger(POSITION_MAGIC) == MagicNumber)
            {
                totalProfit += PositionGetDouble(POSITION_PROFIT);
            }
        }
    }
    return totalProfit;
}

//+------------------------------------------------------------------+
//| Calculate Profit for a Specific Day                              |
//+------------------------------------------------------------------+
double CalculateProfitForDay(datetime startTime, datetime endTime)
{
    double totalProfit = 0;
    HistorySelect(startTime, endTime);
    for(int i = HistoryDealsTotal() - 1; i >= 0; i--)
    {
        ulong ticket = HistoryDealGetTicket(i);
        if(ticket > 0 && HistoryDealGetString(ticket, DEAL_SYMBOL) == _Symbol &&
           HistoryDealGetInteger(ticket, DEAL_MAGIC) == MagicNumber)
        {
            totalProfit += HistoryDealGetDouble(ticket, DEAL_PROFIT);
        }
    }
    return totalProfit;
}


//+------------------------------------------------------------------+
//| Calculate Weekly Profit                                          |
//+------------------------------------------------------------------+
double CalculateWeeklyProfit()
{
    datetime startOfWeek = iTime(_Symbol, PERIOD_W1, 0);
    return CalculateProfitForDay(startOfWeek, TimeCurrent());
}


//+------------------------------------------------------------------+
//| Open Buy Trade Function                                          |
//+------------------------------------------------------------------+
void OpenBuyTrade()
{
    g_buyTradeCount++;
    g_kTradeCount++;
    
    if(g_buyTradeCount >= MaxTrades)
    {
        DebugPrint("Cannot open buy trade. Max trades reached.");
        return;
    }

    double lotSize;
    if(g_lossesCount >= nLosses){
        g_currentLotSize *= Martingale;
        lotSize = g_currentLotSize;
        g_lossesCount = 0;
     } else { 
        lotSize = g_currentLotSize;
     }

    double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
    trade.SetExpertMagicNumber(MagicNumber);

    if(trade.Buy(lotSize, _Symbol, ask, 0, 0, "Buy Trade"))
    {
        g_isBuyActive = true;
        DebugPrint(StringFormat("Buy Trade Opened. Lot Size: %f, Buy Count: %d", lotSize, g_buyTradeCount));
    }
    else
    {
        DebugPrint(StringFormat("Failed to open buy trade. Error: %d", GetLastError()));
    }
}

//+------------------------------------------------------------------+
//| Open Sell Trade Function                                         |
//+------------------------------------------------------------------+
void OpenSellTrade()
{
    g_sellTradeCount++;
    g_kTradeCount++;
    
    if(g_sellTradeCount >= MaxTrades)
    {
        DebugPrint("Cannot open sell trade. Max trades reached.");
        return;
    }

    double lotSize;
    if(g_lossesCount >= nLosses){
        g_currentLotSize *= Martingale;
        lotSize = g_currentLotSize;
        g_lossesCount = 0;
     } else {
        lotSize = g_currentLotSize;
     }
        
    double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    trade.SetExpertMagicNumber(MagicNumber);

    if(trade.Sell(lotSize, _Symbol, bid, 0, 0, "Sell Trade"))
    {
        g_isSellActive = true;
        DebugPrint(StringFormat("Sell Trade Opened. Lot Size: %f, Sell Count: %d", lotSize, g_sellTradeCount));
    }
    else
    {
        DebugPrint(StringFormat("Failed to open sell trade. Error: %d", GetLastError()));
    }
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
    g_totalProfit = CalculateTotalProfit();
        
    if(g_isBuyActive || g_isSellActive)
    {    
       if(g_totalProfit >= g_currentProfitTarget || g_totalProfit <= TotalStopLoss)
       {
           DebugPrint(StringFormat("Profit Target/Stop Loss Reached. Total Profit: %f", g_totalProfit));
           CloseAllTrades();
         // /* 
           g_buyTradeCount = 0;
           g_sellTradeCount = 0;
           g_kTradeCount = 0;
           g_currentLotSize = InitialLotSize;
           g_isBuyActive = false;
           g_isSellActive = false;
           g_accumulatedLoss = 0;
           g_currentProfitTarget = TotalProfitTarget;
           g_totalProfit = 0;
           g_lossesCount = 0;
         //  */
           return;
       }
    }
    
    if(BarsCalculated(g_wprHandle) < 2)
    {
        DebugPrint("Waiting for WPR indicator to calculate sufficient bars...");
        return;
    }
    
    double wprValues[], close[];
    ArraySetAsSeries(wprValues, true);
    ArraySetAsSeries(close, true);
    
    if(CopyBuffer(g_wprHandle, 0, 0, 3, wprValues) <= 0 ||
       CopyClose(_Symbol, PERIOD_CURRENT, 0, 3, close) <= 0)
    {
        Print("Failed to copy data. Error: ", GetLastError());
        return;
    }
    
    DrawChartInformation(); 
    
    if(!IsNewBar())
        return;
        
    // Check for WPR crosses
    bool buySignal = (wprValues[1] > -30 && wprValues[2] <= -30);
    bool sellSignal = (wprValues[1] < -70 && wprValues[2] >= -70);
      
    // Buy Trade Management
    if(!g_isBuyActive && !g_isSellActive && buySignal && BuySignal)
    {
        OpenBuyTrade();
    }
    else if(g_isBuyActive && sellSignal)
    {
       CloseAllTrades();
       
        if(g_totalProfit > 0) {
            if(g_totalProfit > g_accumulatedLoss) {
               g_buyTradeCount = 0;
               g_sellTradeCount = 0;
               g_kTradeCount = 0;
               g_currentLotSize = InitialLotSize;
               g_accumulatedLoss = 0;
               g_currentProfitTarget = TotalProfitTarget;
               g_totalProfit = 0;
               g_lossesCount = 0;
            } else {
               g_accumulatedLoss -=  MathAbs(g_totalProfit);
               g_currentProfitTarget = TotalProfitTarget + g_accumulatedLoss; 
            }
            
        } else if(g_totalProfit < 0) {
         g_lossesCount++;
         g_accumulatedLoss += MathAbs(g_totalProfit);
         g_currentProfitTarget = TotalProfitTarget + g_accumulatedLoss;
        } 
        
        g_totalProfit = 0;
        
        OpenSellTrade();
        
        g_isBuyActive = false;
    }
    
    // Sell Trade Management
    if(!g_isSellActive && !g_isBuyActive && sellSignal && SellSignal)
    {
        OpenSellTrade();
    }
    else if(g_isSellActive && buySignal)
    {
        CloseAllTrades();
            
         if(g_totalProfit > 0) {
            if(g_totalProfit > g_accumulatedLoss) {
               g_buyTradeCount = 0;
               g_sellTradeCount = 0;
               g_kTradeCount = 0;
               g_currentLotSize = InitialLotSize;
               g_accumulatedLoss = 0;
               g_currentProfitTarget = TotalProfitTarget;
               g_totalProfit = 0;
               g_lossesCount = 0;
            } else {
               g_accumulatedLoss -=  MathAbs(g_totalProfit);
               g_currentProfitTarget = TotalProfitTarget + g_accumulatedLoss; 
            }
            
        } else if(g_totalProfit < 0) {
         g_lossesCount++;
         g_accumulatedLoss += MathAbs(g_totalProfit);
         g_currentProfitTarget = TotalProfitTarget + g_accumulatedLoss;
        } 
        
        g_totalProfit = 0;
        
        if(BuySignal) OpenBuyTrade();
        
        g_isSellActive = false;
    }
}

//+------------------------------------------------------------------+
//| Draw Chart Information                                           |
//+------------------------------------------------------------------+

    string objectName3 = "ChartInfo3";
    string objectName4 = "ChartInfo4";

    double profitWeek = CalculateWeeklyProfit();

//+------------------------------------------------------------------+
//| Expert deinitialization function                                 |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    CloseAllTrades();
    ObjectDelete(0, "ChartInfo");
    ObjectDelete(0, "ChartInfo2");
    ObjectDelete(0, "ChartInfo3");
    ObjectDelete(0, "ChartInfo4");
    ObjectDelete(0, "ChartInfo5");
    
    if(g_wprHandle != INVALID_HANDLE)
        IndicatorRelease(g_wprHandle);
    
    Print("EA deinitialized. Reason: ", reason);
    
    DebugPrint(StringFormat("EA Deinitialized. Reason: %d", reason));
}
